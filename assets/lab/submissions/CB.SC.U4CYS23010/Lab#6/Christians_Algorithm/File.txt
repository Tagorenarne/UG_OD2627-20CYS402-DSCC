1ST EXPERIMENT:
NO CHANGE IN FILE JUST RUN:
	sudo docker-compose up --build


# Ensure any background containers from previous runs are removed
	sudo docker-compose down

# Modify docker-compose.yml to set the artificial delay to 0ms
	sed -i 's/- ARTIFICIAL_DELAY_MS=.*/- ARTIFICIAL_DELAY_MS=0/' docker-compose.yml

	sudo docker-compose up --build
